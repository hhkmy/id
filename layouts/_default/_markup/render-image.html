{{/* layouts/_default/_markup/render-image.html */}} {{- $img := .Page.Resources.GetMatch .Destination -}} {{- $alt := .PlainText | safeHTML -}} {{- $title := .Title | safeHTML -}} {{- if $img -}} {{/* Define responsive breakpoints */}} {{- $widths := slice 480 768 1024 1280 1600 -}} {{- $srcsetAvif := slice -}} {{- $srcsetWebp := slice -}} {{- $srcsetOriginal := slice -}} {{/* Generate responsive formats */}} {{- range $widths -}} {{- if le . $img.Width -}} {{- $avif := ($img.Resize (printf "%dx avif" .)) -}} {{- $webp := ($img.Resize (printf "%dx webp" .)) -}} {{- $original := ($img.Resize (printf "%dx" .)) -}} {{- $srcsetAvif = $srcsetAvif | append (printf "%s %dw" $avif.RelPermalink .) -}} {{- $srcsetWebp = $srcsetWebp | append (printf "%s %dw" $webp.RelPermalink .) -}} {{- $srcsetOriginal = $srcsetOriginal | append (printf "%s %dw" $original.RelPermalink .) -}} {{- end -}} {{- end -}} {{/* Fallback if image is smaller than smallest width */}} {{- if not $srcsetAvif -}} {{- $avif := ($img.Resize "avif") -}} {{- $webp := ($img.Resize "webp") -}} {{- $srcsetAvif = slice (printf "%s" $avif.RelPermalink) -}} {{- $srcsetWebp = slice (printf "%s" $webp.RelPermalink) -}} {{- $srcsetOriginal = slice (printf "%s" $img.RelPermalink) -}} {{- end -}} {{- else -}} {{- end -}}
